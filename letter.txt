  <script>
    // 你的素材路径（按你仓库现有文件名）
    const ASSETS = {
      base: "./assets/feng.png",
      flap: "./assets/gai.png",
      stamp: "./assets/ko.png",
      paper: "./assets/生日信.png",
    };

    const cover = document.getElementById("cover");
    const reader = document.getElementById("reader");
    const envelope = document.getElementById("envelope");

    const envBaseImg = document.getElementById("envBaseImg");
    const envFlapImg = document.getElementById("envFlapImg");
    const stampImg = document.getElementById("stampImg");
    const paperPreviewImg = document.getElementById("paperPreviewImg");
    const paperImg = document.getElementById("paperImg");

    // 绑定图片
    envBaseImg.src = ASSETS.base;
    envFlapImg.src = ASSETS.flap;
    stampImg.src = ASSETS.stamp;
    paperPreviewImg.src = ASSETS.paper;
    paperImg.src = ASSETS.paper;

    // 预加载，减少首次点击空白
    Object.values(ASSETS).forEach(src => {
      const img = new Image();
      img.src = src;
    });

    function openLetter() {
      envelope.classList.add("opened");
      const total = 900; // 动画主时长（ms）
      setTimeout(() => {
        cover.style.display = "none";
        reader.classList.add("show");
        document.getElementById("scrollArea").scrollTo({ top: 0, behavior: "instant" });
      }, total);
    }

    function closeLetter() {
      reader.classList.remove("show");
      cover.style.display = "grid";
      envelope.classList.remove("opened");
      // 把邮戳恢复回来（方便你测试多次）
      document.getElementById("stampBtn").classList.remove("gone");
    }

    document.getElementById("stampBtn").addEventListener("click", () => {
      const btn = document.getElementById("stampBtn");
      btn.classList.add("gone");  // 邮戳消失
      openLetter();               // 开信
    });

    document.getElementById("backBtn").addEventListener("click", closeLetter);
    document.getElementById("toTopBtn").addEventListener("click", () => {
      document.getElementById("scrollArea").scrollTo({ top: 0, behavior: "smooth" });
    });
  </script>
